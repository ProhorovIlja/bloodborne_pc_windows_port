cmake_minimum_required(VERSION 3.20)
project(sh_v20_range_guard LANGUAGES CXX)
include(CTest)
add_executable(test_map_range_io test_map_range_io.cpp)
target_compile_features(test_map_range_io PRIVATE cxx_std_20)

 target_compile_options(test_map_range_io PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME map_range_io COMMAND test_map_range_io)
