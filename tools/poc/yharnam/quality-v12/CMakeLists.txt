cmake_minimum_required(VERSION 3.20)
project(seamless_hunt_quality_policies LANGUAGES CXX)
include(CTest)
add_executable(test_quality test_quality.cpp)
target_compile_features(test_quality PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(test_quality PRIVATE /W4 /WX)
else()
  target_compile_options(test_quality PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_test(NAME quality_filters COMMAND test_quality)
