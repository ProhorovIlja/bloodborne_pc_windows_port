cmake_minimum_required(VERSION 3.20)
project(sh_spectator_v13 LANGUAGES C CXX)
include(CTest)
add_executable(test_spectator test_spectator.cpp)
target_compile_features(test_spectator PRIVATE cxx_std_20)
add_executable(test_camera test_camera.c)
set_target_properties(test_camera PROPERTIES C_STANDARD 11)
if(MSVC)
 target_compile_options(test_spectator PRIVATE /W4 /WX)
 target_compile_options(test_camera PRIVATE /W4 /WX)
else()
 target_compile_options(test_spectator PRIVATE -Wall -Wextra -Werror -Wpedantic)
 target_compile_options(test_camera PRIVATE -Wall -Wextra -Werror -Wpedantic)
 target_link_libraries(test_camera PRIVATE m)
endif()
add_test(NAME spectator_policy COMMAND test_spectator)
add_test(NAME spectator_camera COMMAND test_camera)
