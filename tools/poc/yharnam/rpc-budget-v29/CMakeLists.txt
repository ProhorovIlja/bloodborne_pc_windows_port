cmake_minimum_required(VERSION 3.20)
project(sh_v29_rpc_budget LANGUAGES CXX)
include(CTest)
add_executable(test_rpc_budget test_rpc_budget.cpp)
target_compile_features(test_rpc_budget PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(test_rpc_budget PRIVATE /W4 /WX)
else()
 target_compile_options(test_rpc_budget PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME rpc_budget COMMAND test_rpc_budget)
