cmake_minimum_required(VERSION 3.20)
project(sh_v30_auth_revocation LANGUAGES CXX)
include(CTest)
add_executable(test_auth_revocation test_auth_revocation.cpp)
target_compile_features(test_auth_revocation PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(test_auth_revocation PRIVATE /W4 /WX)
else()
 target_compile_options(test_auth_revocation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME auth_revocation COMMAND test_auth_revocation)
