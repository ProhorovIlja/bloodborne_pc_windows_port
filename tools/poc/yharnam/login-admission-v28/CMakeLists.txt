cmake_minimum_required(VERSION 3.20)
project(sh_v28_login_admission LANGUAGES CXX)
include(CTest)
add_executable(test_login_admission test_login_admission.cpp)
target_compile_features(test_login_admission PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(test_login_admission PRIVATE /W4 /WX)
else()
  target_compile_options(test_login_admission PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME login_admission COMMAND test_login_admission)
