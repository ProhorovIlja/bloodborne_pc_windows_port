cmake_minimum_required(VERSION 3.20)
project(sh_v25_member_identity LANGUAGES CXX)
include(CTest)
add_executable(test_member_identity test_member_identity.cpp)
target_compile_features(test_member_identity PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(test_member_identity PRIVATE /W4 /WX)
else()
  target_compile_options(test_member_identity PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME member_identity COMMAND test_member_identity)
