cmake_minimum_required(VERSION 3.20)
project(sh_v24_stream_bounds LANGUAGES CXX)
include(CTest)
add_executable(test_stream_bounds test_stream_bounds.cpp)
target_compile_features(test_stream_bounds PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(test_stream_bounds PRIVATE /W4 /WX)
else()
 target_compile_options(test_stream_bounds PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME stream_bounds COMMAND test_stream_bounds)
