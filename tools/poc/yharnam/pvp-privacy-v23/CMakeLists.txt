cmake_minimum_required(VERSION 3.20)
project(sh_v23_pvp_privacy LANGUAGES CXX)
include(CTest)
add_executable(test_pvp_privacy test_pvp_privacy.cpp)
target_compile_features(test_pvp_privacy PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(test_pvp_privacy PRIVATE /W4 /WX)
else()
 target_compile_options(test_pvp_privacy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME pvp_privacy COMMAND test_pvp_privacy)
