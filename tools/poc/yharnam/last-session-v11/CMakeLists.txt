cmake_minimum_required(VERSION 3.20)
project(seamless_hunt_last_session LANGUAGES CXX)
include(CTest)
add_executable(test_last_session test_last_session.cpp)
target_compile_features(test_last_session PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(test_last_session PRIVATE /W4 /WX)
else()
 target_compile_options(test_last_session PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_test(NAME last_session_policy COMMAND test_last_session)
