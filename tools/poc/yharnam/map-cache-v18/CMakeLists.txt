cmake_minimum_required(VERSION 3.20)
project(sh_v18_map_cache LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
add_executable(test_map_cache test_map_cache.cpp)
target_compile_features(test_map_cache PRIVATE cxx_std_20)
target_link_libraries(test_map_cache PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(test_map_cache PRIVATE /W4 /WX)
else()
 target_compile_options(test_map_cache PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME map_cache COMMAND test_map_cache)
