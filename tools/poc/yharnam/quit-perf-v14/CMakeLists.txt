cmake_minimum_required(VERSION 3.20)
project(sh_v14_performance_monitor LANGUAGES CXX)
include(CTest)
add_executable(test_perf_watch test_perf_watch.cpp)
target_compile_features(test_perf_watch PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(test_perf_watch PRIVATE /W4 /WX)
else()
 target_compile_options(test_perf_watch PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_test(NAME perf_watch COMMAND test_perf_watch)
