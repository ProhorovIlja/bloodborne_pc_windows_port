cmake_minimum_required(VERSION 3.20)
project(seamless_hunt_native_channel_guard LANGUAGES CXX)
include(CTest)
add_executable(test_native_direct_filter test_direct_filter.cpp)
target_compile_features(test_native_direct_filter PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(test_native_direct_filter PRIVATE /W4 /WX)
else()
  target_compile_options(test_native_direct_filter PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_test(NAME native_direct_channel_guard COMMAND test_native_direct_filter)

find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(test_native_guard_live test_guard_live.cpp)
target_compile_features(test_native_guard_live PRIVATE cxx_std_20)
target_link_libraries(test_native_guard_live PRIVATE OpenSSL::Crypto)
if(MSVC)
  target_compile_options(test_native_guard_live PRIVATE /W4 /WX)
else()
  target_compile_options(test_native_guard_live PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_test(NAME native_guard_live COMMAND test_native_guard_live)
