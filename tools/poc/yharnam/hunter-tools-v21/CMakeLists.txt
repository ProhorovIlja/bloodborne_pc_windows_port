cmake_minimum_required(VERSION 3.20)
project(seamless_hunter_tools LANGUAGES CXX)
include(CTest)
add_executable(test_hunter_tools_policy test_hunter_tools_policy.cpp)
target_compile_features(test_hunter_tools_policy PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(test_hunter_tools_policy PRIVATE /W4 /WX)
else()
  target_compile_options(test_hunter_tools_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME hunter_tools_policy COMMAND test_hunter_tools_policy)
