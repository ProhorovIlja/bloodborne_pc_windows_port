cmake_minimum_required(VERSION 3.20)
project(sh_v22_server_guard LANGUAGES CXX)
include(CTest)
add_executable(test_server_guard test_server_guard.cpp)
target_compile_features(test_server_guard PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(test_server_guard PRIVATE /W4 /WX)
else()
 target_compile_options(test_server_guard PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME server_guard COMMAND test_server_guard)
