cmake_minimum_required(VERSION 3.20)
project(sh_v26_ingress_quota LANGUAGES CXX)
include(CTest)
add_executable(test_ingress_quota test_ingress_quota.cpp)
target_compile_features(test_ingress_quota PRIVATE cxx_std_20)

 target_compile_options(test_ingress_quota PRIVATE /W4 /WX)
else()
 target_compile_options(test_ingress_quota PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ingress_quota COMMAND test_ingress_quota)
