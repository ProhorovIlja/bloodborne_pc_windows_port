cmake_minimum_required(VERSION 3.20)
project(shyt_native_game_asset_reader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ZLIB REQUIRED)
add_library(bb_native_assets STATIC src/asset_archive.cpp)
target_include_directories(bb_native_assets PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(bb_native_assets PUBLIC ZLIB::ZLIB)
if(MSVC)
 target_compile_options(bb_native_assets PRIVATE /W4 /WX)
else()
 target_compile_options(bb_native_assets PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(bb-native-assets src/main.cpp)
target_link_libraries(bb-native-assets PRIVATE bb_native_assets)
enable_testing()
add_executable(test-native-assets tests/test_assets.cpp)
target_link_libraries(test-native-assets PRIVATE bb_native_assets)
add_test(NAME native_source_port_assets COMMAND test-native-assets)
